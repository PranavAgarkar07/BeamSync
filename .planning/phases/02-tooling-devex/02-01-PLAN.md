### Task 2.1.4 — GitHub Actions CI

Create `.github/workflows/ci.yml`:
```yaml
name: CI
on: [push, pull_request]
jobs:
  lint-go:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-go@v5
        with: { go-version: '1.25' }
      - run: (cd beamsync && golangci-lint run ./...)
      - run: (cd desktop && golangci-lint run ./...)

  lint-svelte:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: '20' }
      - run: cd desktop/frontend && npm ci && npm run lint

  test-go:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-go@v5
        with: { go-version: '1.25' }
      - run: (cd beamsync && go test ./...)
      - run: (cd desktop && go test ./...)

  test-svelte:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: '20' }
      - run: cd desktop/frontend && npm ci && npm run test

  build:
    runs-on: ubuntu-latest
    needs: [lint-go, lint-svelte, test-go, test-svelte]
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-go@v5
        with: { go-version: '1.25' }
      - uses: actions/setup-node@v4
        with: { node-version: '20' }
      - run: cd desktop/frontend && npm ci && npm run build
      - run: (cd beamsync && go build ./...)
      - run: (cd desktop && go build ./...)
```

<acceptance_criteria>
- `.github/workflows/ci.yml` exists and is valid YAML
- Workflow covers lint (Go + Svelte), test (Go + Svelte), and build
</acceptance_criteria>
